{{- range (index .Values "opendkim-server" "domains") }}

    cat /etc/dkimkeys/{{ .domain }}/{{ .selector | default "mail" }}.txt
  # -> TXT  {{ .selector | default "mail" }}._domainkey.{{ .domain }}
{{- end }}
